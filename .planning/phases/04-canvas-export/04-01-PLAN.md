// === DOWNLOAD: Resolution picker and dropdown logic ===

// Toggle dropdown visibility when arrow button clicked
const arrowBtn = document.getElementById('download-arrow-btn');
const dropdown = document.getElementById('resolution-dropdown');

arrowBtn.addEventListener('click', function(event) {
  event.stopPropagation(); // Prevent outside click handler from immediately closing
  
  const isOpen = dropdown.style.display === 'block';
  
  if (isOpen) {
    // Close dropdown
    dropdown.style.display = 'none';
    arrowBtn.setAttribute('aria-expanded', 'false');
  } else {
    // Open dropdown
    dropdown.style.display = 'block';
    arrowBtn.setAttribute('aria-expanded', 'true');
  }
});

// Close dropdown when clicking outside
document.addEventListener('click', function(event) {
  const container = document.querySelector('.split-button-container');
  
  // If click is outside split-button container and dropdown is open, close it
  if (!container.contains(event.target) && dropdown.style.display === 'block') {
    dropdown.style.display = 'none';
    arrowBtn.setAttribute('aria-expanded', 'false');
  }
});

// Close dropdown on Escape key
document.addEventListener('keydown', function(event) {
  if (event.key === 'Escape' && dropdown.style.display === 'block') {
    dropdown.style.display = 'none';
    arrowBtn.setAttribute('aria-expanded', 'false');
  }
});

// Handle resolution selection
const dropdownItems = document.querySelectorAll('.dropdown-item');

dropdownItems.forEach(function(item) {
  item.addEventListener('click', function() {
    const resolution = parseInt(item.getAttribute('data-resolution'), 10); // 1 or 2
    
    // Update state
    state.selectedResolution = resolution;
    
    // Update main button label
    const downloadBtn = document.getElementById('download-btn');
    downloadBtn.textContent = 'Download (' + resolution + '×)';
    
    // Update checkmark visibility (show on selected, hide on others)
    dropdownItems.forEach(function(otherItem) {
      const isSelected = otherItem === item;
      otherItem.setAttribute('aria-checked', isSelected ? 'true' : 'false');
      
      // Toggle checkmark visibility
      const checkmark = otherItem.querySelector('.checkmark');
      checkmark.style.visibility = isSelected ? 'visible' : 'hidden';
      
      // Toggle selected class for styling (if needed)
      if (isSelected) {
        otherItem.classList.add('selected');
      } else {
        otherItem.classList.remove('selected');
      }
    });
    
    // Close dropdown after selection
    dropdown.style.display = 'none';
    arrowBtn.setAttribute('aria-expanded', 'false');
  });
});
```

// Enable/disable download button based on whether photo is uploaded
function updateDownloadState() {
  const downloadBtn = document.getElementById('download-btn');
  const arrowBtn = document.getElementById('download-arrow-btn');
  const hasPhoto = state.photoImage !== null;
  
  // Enable both buttons if photo exists, disable if no photo

// === DOWNLOAD: Export handler ===

const downloadBtn = document.getElementById('download-btn');

downloadBtn.addEventListener('click', function() {
  // Guard: should never fire when disabled, but double-check
  if (!state.photoImage) {
    console.warn('Download attempted with no photo loaded');
    return;
  }
  
  // Get selected resolution (1 or 2)
  const pixelRatio = state.selectedResolution;
  
  // Export stage to PNG data URL at selected resolution
  const dataURL = stage.toDataURL({
    pixelRatio: pixelRatio,
    mimeType: 'image/png',
    quality: 1
  });
  
  // Create temporary <a> element to trigger download
  const link = document.createElement('a');
  link.download = 'profile.png'; // Fixed filename per D-06
  link.href = dataURL;
  
  // Trigger download (click and remove)
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  
  console.log('Downloaded profile.png at ' + pixelRatio + '× resolution');
});
```
